function inline(el,s){                   // **bold**, everything else as plain text
  for(const part of s.split(/(\*\*[^*]+\*\*)/)){
    if(/^\*\*[^*]+\*\*$/.test(part)){const b=document.createElement('strong');b.textContent=part.slice(2,-2);el.appendChild(b);}
    else if(part) el.appendChild(document.createTextNode(part));
  }
}
function setMsg(el,text){
  el.replaceChildren(); let ul=null;
  for(const line of text.split('\n')){
    const m=line.match(/^\s*[-•*]\s+(.*)/);
    if(m){ if(!ul){ul=document.createElement('ul');el.appendChild(ul);}
           const li=document.createElement('li');inline(li,m[1]);ul.appendChild(li);continue; }
    ul=null; if(!line.trim())continue;
    const p=document.createElement('p');inline(p,line.replace(/^#+\s*/,''));el.appendChild(p);
  }
}
